                         worker.title.toLowerCase().includes(searchQuery.toLowerCase()) ||
                         worker.skills.some(s => s.toLowerCase().includes(searchQuery.toLowerCase()));

      const matchDist = worker.distanceKm <= maxDistance;
      return matchCat && matchQuery && matchVerified && matchDist;
    }).sort((a, b) => {
      if (sortOption === 'rating') return b.rating - a.rating;
      if (sortOption === 'price_low') return a.startingPrice - b.startingPrice;
      if (sortOption === 'distance') return a.distanceKm - b.distanceKm;
      return b.jobsCompleted - a.jobsCompleted; // recommended
    });

      selectedWorker: worker,

              className={`px-3.5 py-2 rounded-xl transition-colors ${activeTab === 'home' ? 'bg-stone-200/70 text-stone-900 font-bold' : 'hover:bg-stone-100'}`}

              Home

            >
              Find Workers
            </button>
            <button 
              onClick={() => setActiveTab('jobs')}
              className={`px-3.5 py-2 rounded-xl transition-colors ${activeTab === 'jobs' ? 'bg-stone-200/70 text-stone-900 font-bold' : 'hover:bg-stone-100'}`}

            setActiveTab={setActiveTab}

          />

          <WorkerProfileView

            handleStartBooking={handleStartBooking}
            showToast={showToast}
